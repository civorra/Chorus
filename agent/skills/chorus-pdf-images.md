### Script template — Images mode (`--images`)

All pages rendered to PNG at 150 DPI via `pdftoppm`, submitted to Claude vision.
Use when the document is mostly diagrams or scanned.
Output: `<NNN>-<slug>-vision.md`

```python
#!/usr/bin/env python3
"""
chorus-pdf extraction script — images mode (--images)
Generated by chorus-pdf skill
Sandbox : <sandbox-name>
Source  : <input-pdf-path>
Output  : <output-md-path>    (e.g. corpus/003-uk-approved-doc-a-2013-vision.md)
Pages   : <total-pages>
DPI     : 150
"""

import sys
import base64
import json
import glob
import re
import time
import tempfile
import subprocess
import urllib.request
import urllib.error
import os

PDF_PATH    = "<input-pdf-path>"
OUTPUT_PATH = "<output-md-path>"
CHUNK_SIZE  = 5
DPI         = 150
MAX_RETRIES = 4
API_KEY     = os.environ.get("ANTHROPIC_API_KEY", "")
API_URL     = "https://api.anthropic.com/v1/messages"

def _log_environment():
    """Log the running interpreter (images mode has no os.execv() bootstrap)."""
    print(f"[chorus-pdf] Environment: (interpreter: {sys.executable})", file=sys.stderr)

_log_environment()

# ── Pre-flight API key check (explicit --images: no silent fallback) ───────
if not API_KEY:
    print("⛔ [chorus-pdf] ANTHROPIC_API_KEY not set — --images requires Claude vision.\n"
          "   Set ANTHROPIC_API_KEY (see ~/.config/anthropic/api_key) or omit --images\n"
          "   to use automatic text-only fallback.",
          file=sys.stderr)
    sys.exit(1)

def _probe_claude(api_key):
    """Validate the API key with a minimal 1-token request before any PDF processing.
    Returns True if the key is valid (or throttled). Returns False on 401/403 or network error.
    """
    import json, urllib.request, urllib.error
    payload = {"model": "claude-haiku-4-5", "max_tokens": 1,
               "messages": [{"role": "user", "content": "ping"}]}
    headers = {"x-api-key": api_key, "anthropic-version": "2023-06-01",
               "content-type": "application/json"}
    try:
        req = urllib.request.Request(
            "https://api.anthropic.com/v1/messages",
            data=json.dumps(payload).encode(), headers=headers, method="POST"
        )
        urllib.request.urlopen(req, timeout=10)
        return True
    except urllib.error.HTTPError as e:
        return e.code in (429, 529)
    except Exception:
        return False

print("[chorus-pdf] ANTHROPIC_API_KEY set — probing Claude availability …", file=sys.stderr)
if not _probe_claude(API_KEY):
    print("⛔ [chorus-pdf] ANTHROPIC_API_KEY is invalid or API unreachable — --images aborted.\n"
          "   Verify your key at https://console.anthropic.com or check network connectivity.",
          file=sys.stderr)
    sys.exit(1)
print("[chorus-pdf] ANTHROPIC_API_KEY ✅ — Claude available, images mode activated.", file=sys.stderr)

PROMPT = """<verbatim vision extraction prompt — see Phase 2>"""


def render_pages(pdf_path, tmpdir):
    """Render all PDF pages to PNG files using pdftoppm."""
    prefix = os.path.join(tmpdir, "page")
    result = subprocess.run(
        ["pdftoppm", "-r", str(DPI), "-png", pdf_path, prefix],
        capture_output=True
    )
    if result.returncode != 0:
        raise RuntimeError(f"pdftoppm failed:\n{result.stderr.decode()}")
    pages = sorted(glob.glob(os.path.join(tmpdir, "page-*.png")))
    if not pages:
        pages = sorted(glob.glob(os.path.join(tmpdir, "page.*.png")))
    if not pages:
        raise RuntimeError("pdftoppm produced no PNG files — is the PDF password-protected?")
    return pages


# → image_to_b64() — see "Shared utility — Auto & Images modes" above


def call_claude(page_paths, page_offset):
    """Send a chunk of page PNGs to Claude vision."""
    content = []
    for i, path in enumerate(page_paths):
        page_num = page_offset + i + 1
        content.append({
            "type": "text",
            "text": f"[Processing page {page_num}]\n\n{PROMPT}"
        })
        content.append({
            "type": "image",
            "source": {
                "type": "base64",
                "media_type": "image/png",
                "data": image_to_b64(path)
            }
        })

    payload = {
        "model": "claude-opus-4-5",
        "max_tokens": 8192,
        "messages": [{"role": "user", "content": content}]
    }
    headers = {
        "x-api-key": API_KEY,
        "anthropic-version": "2023-06-01",
        "content-type": "application/json"
    }
    label = f"{page_offset+1}-{page_offset+len(page_paths)}"
    return call_claude_with_retry(payload, headers, label)
    # → call_claude_with_retry: see "Shared utilities — Text & Hybrid modes" above


def main():
    if not API_KEY:
        print("⛔ ANTHROPIC_API_KEY not set", file=sys.stderr)
        sys.exit(1)

    r = subprocess.run(["pdftoppm", "--help"], capture_output=True)
    if r.returncode not in (0, 99):
        print("⛔ pdftoppm not found. Install: sudo apt install poppler-utils", file=sys.stderr)
        sys.exit(1)

    with tempfile.TemporaryDirectory(prefix="chorus-pdf-") as tmpdir:
        print(f"[chorus-pdf] Rendering pages at {DPI} DPI ...", file=sys.stderr)
        pages  = render_pages(PDF_PATH, tmpdir)
        total  = len(pages)
        chunks = [pages[i:i+CHUNK_SIZE] for i in range(0, total, CHUNK_SIZE)]
        print(f"[chorus-pdf] {total} pages → {len(chunks)} chunk(s) of {CHUNK_SIZE}",
              file=sys.stderr)

        all_text = []
        for idx, chunk in enumerate(chunks):
            start = idx * CHUNK_SIZE
            label = f"{start+1}-{start+len(chunk)}"
            print(f"[chorus-pdf] Chunk {idx+1}/{len(chunks)} (pages {label}) ...",
                  file=sys.stderr)
            text = call_claude(chunk, start)
            all_text.append(text)
            print(f"[chorus-pdf]   → {len(text)} chars", file=sys.stderr)

    with open(OUTPUT_PATH, "w", encoding="utf-8") as f:
        f.write(f"# ORIGINAL: {PDF_PATH}\n\n" + "\n\n".join(all_text))
    print(f"[chorus-pdf] Vision extraction: Claude active ({total} page(s) described via API)",
          file=sys.stderr)
    print(f"[chorus-pdf] ✅ Written to {OUTPUT_PATH}", file=sys.stderr)


if __name__ == "__main__":
    main()
```
