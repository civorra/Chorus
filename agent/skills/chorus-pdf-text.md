### detect_and_tag_formulas — Text mode utility

> Moved here from Shared utilities (ENG-042): only called in Text mode.

def detect_and_tag_formulas(text):
    """Detect likely mathematical-formula fragments in a page's assembled text
    and wrap them with a `[FORMULA: ...]` placeholder, mirroring the existing
    `[FIGURE — not extracted]` convention (see § "Formula strategy").

    Motivated by audit [#J] § F1/F2/F3: unlike figures, formulas were never
    flagged in text mode — ambiguous exponents (`2^128` → `2128`) are
    silently indistinguishable from plain integers, and no signal exists for
    a human/operator to locate risk zones without reading the whole document.

    This is a best-effort, line-local heuristic — no LaTeX transcription is
    attempted beyond a few trivial, unambiguous cases (superscript Unicode
    digits). Returns the text with `[FORMULA: ...]` markers inserted inline,
    immediately after each detected fragment.
    """
    import re

    # Unicode math symbols whose mere presence strongly suggests a formula
    MATH_SYMBOLS = re.compile(r'[≥≤⊕√∑∏²³ℓλσ±×÷≈≠∈∉⊂⊃∀∃∞]')

    # Functional notation: f(x, y) / XH(M, ℓ) style — identifier immediately
    # followed by a parenthesised argument list containing at least one
    # comma or a known math-symbol argument.
    FUNC_NOTATION = re.compile(r'\b[A-Zℓλσ][A-Za-z0-9]{0,4}\([^()]{1,40},[^()]{0,40}\)')

    # Ambiguous flattened exponent: a leading "2" directly followed by 2-3
    # more digits, only flagged when a crypto/security-flavoured keyword is
    # present nearby (same line or same paragraph) to limit false positives
    # on years/identifiers (audit [#J] § F1).
    EXPONENT_CONTEXT = re.compile(
        r'(?i)\b(bits?|calculs?|op[ée]rations?|complexit[ée]s?|collisions?|operations?|complexity)\b'
    )
    EXPONENT_PATTERN = re.compile(r'\b2\d{2,3}\b')

    def tag_line(line):
        tags = []

        if FUNC_NOTATION.search(line):
            tags.append("functional notation")
        if MATH_SYMBOLS.search(line):
            tags.append("math symbol")
        if EXPONENT_PATTERN.search(line) and EXPONENT_CONTEXT.search(line):
            tags.append("ambiguous exponent")

        if not tags:
            return line

        # Best-effort LaTeX transcription — only for the unambiguous
        # superscript-Unicode case; everything else falls back to a bare
        # placeholder (never invent a transcription we cannot verify).
        latex_hint = ""
        m = re.search(r'(\w)([²³])', line)
        if m:
            base, exp = m.group(1), m.group(2)
            power = {"²": "2", "³": "3"}[exp]
            latex_hint = f" — best-effort LaTeX: {base}^{power}"

        reason = ", ".join(tags)
        return f"{line}\n[FORMULA: pattern detected ({reason}){latex_hint} — verify against source]"

    return "\n".join(tag_line(line) for line in text.split("\n"))


def extract_page_via_pdftotext_layout(pdf_path, page_num):
    """§1.4c/§1.4d — Fallback extraction for borderless or rotated-header tables.

    Uses pdftotext -layout (Poppler) which preserves the absolute X/Y grid of
    the page — a completely independent layout engine from pdfminer/pdfplumber.
    Returns the raw layout-preserving text for the page, or None on failure.

    Required: poppler-utils installed (sudo apt install poppler-utils).
    Cost: fully offline, no API call.
    """
    import subprocess
    try:
        result = subprocess.run(
            ["pdftotext", "-f", str(page_num), "-l", str(page_num), "-layout",
             pdf_path, "-"],
            capture_output=True, text=True, timeout=30
        )
        if result.returncode != 0:
            return None
        return result.stdout
    except FileNotFoundError:
        print(f"[chorus-pdf] ⚠️  pdftotext not found (poppler-utils) — "
              "borderless table fallback disabled. Install: sudo apt install poppler-utils",
              file=sys.stderr)
        return None
    except Exception:
        return None



### Script template — Text mode (no flag + Claude unavailable, or forced text mode)

Uses `pdfminer.six` only. No API key, no network. Figures produce a placeholder.
Output: `<NNN>-<slug>-text.md`

```python
#!/usr/bin/env python3
"""
chorus-pdf extraction script — text mode (default)
Generated by chorus-pdf skill
Sandbox : <sandbox-name>
Source  : <input-pdf-path>
Output  : <output-md-path>    (e.g. corpus/003-uk-approved-doc-a-2013-text.md)
"""

import sys
import os

# --- pdfplumber interpreter bootstrap ---
# If pdfplumber is not importable by the current interpreter, re-exec transparently
# with the dedicated pipx venv where it is installed.
try:
    import pdfplumber as _chk; del _chk
except ImportError:
    _pipx_py = os.path.expanduser("~/.local/share/pipx/venvs/pdfplumber/bin/python3")
    if os.path.exists(_pipx_py) and sys.executable != _pipx_py:
        os.execv(_pipx_py, [_pipx_py] + sys.argv)
    # else: pdfplumber genuinely absent — graceful fallback will apply later

def _log_environment():
    """Log the actually-running interpreter and library versions, emitted AFTER
    the os.execv() bootstrap above so it reflects the final interpreter, not the
    one the script was originally launched with."""
    import importlib.metadata as _ilmd
    parts = []
    for pkg in ("pdfminer.six", "pdfplumber"):
        try:
            parts.append(f"{pkg}=={_ilmd.version(pkg)}")
        except _ilmd.PackageNotFoundError:
            parts.append(f"{pkg}==<not installed>")
    print(f"[chorus-pdf] Environment: {', '.join(parts)} (interpreter: {sys.executable})",
          file=sys.stderr)

_log_environment()

PDF_PATH    = "<input-pdf-path>"
OUTPUT_PATH = "<output-md-path>"

FIGURE_PLACEHOLDER = (
    "[FIGURE — not extracted]\n"
    "[Run chorus-pdf with --hybrid, --auto or --images to extract figures via LLM vision]"
)


# → analyse_pages() / detect_table_columns() / extract_tables_from_page()
#   see "Shared utilities — Text & Hybrid modes" above


def main():
    # Optional pdfplumber for vector table reconstruction
    try:
        import pdfplumber as pdfplumber_mod
        HAS_PDFPLUMBER = True
    except ImportError:
        HAS_PDFPLUMBER = False
        print("[chorus-pdf] ⚠️  pdfplumber not available — vector tables will not be reconstructed",
              file=sys.stderr)
    # → detect_and_tag_formulas() — see "Shared utilities — Text & Hybrid modes" above

    print(f"[chorus-pdf] Text mode — {PDF_PATH}", file=sys.stderr)
    page_data = analyse_pages(PDF_PATH)   # → "Shared utilities — Text & Hybrid modes" above
    plumber_pdf = pdfplumber_mod.open(PDF_PATH) if HAS_PDFPLUMBER else None

    parts = []
    # Provenance header — enables chorus-check (Phase E0.b) to resolve the original
    # document without basename-matching heuristics. See chorus-check.md § Phase E0.b.
    parts.append(f"# ORIGINAL: {PDF_PATH}")
    fig_pages   = 0
    total_tables = 0

    for page_num in sorted(page_data):
        pdata = page_data[page_num]
        blocks, has_figure, has_table, page_height = \
            pdata['texts'], bool(pdata['figures']), pdata['has_table'], pdata['height']
        elements = []  # (y_center, content)

        # Vector table reconstruction
        table_bboxes = []  # (tb_x0, tb_y_min, tb_x1, tb_y_max) in pdfminer coords
        if plumber_pdf and has_table:
            table_entries = extract_tables_from_page(plumber_pdf.pages[page_num - 1])
            for (tx0, t_top, tx1, t_bottom), md in table_entries:
                y_min = page_height - t_bottom
                y_max = page_height - t_top
                y_c   = (y_min + y_max) / 2
                table_bboxes.append((tx0, y_min, tx1, y_max))
                elements.append((y_c, md))
                total_tables += 1

        # Text blocks — suppressed only if inside a table bbox (2D: X center + Y).
        # Y-only suppression causes false deduplication when a table occupies only
        # part of the page width (e.g. a risk matrix on the right while text flows
        # on the left at the same Y coordinates).
        for (text, y_center, x0, x1) in blocks:
            x_center = (x0 + x1) / 2
            in_table = any(
                tb_x0 - 5 <= x_center <= tb_x1 + 5 and tb_y_min <= y_center <= tb_y_max
                for (tb_x0, tb_y_min, tb_x1, tb_y_max) in table_bboxes
            )
            if in_table:
                continue
            elements.append((y_center, text))

        # Figure placeholder
        if has_figure:
            elements.append((0, FIGURE_PLACEHOLDER))
            fig_pages += 1

        elements.sort(key=lambda e: e[0], reverse=True)
        body = "\n\n".join(c for (_, c) in elements)
        parts.append(f"=== PAGE {page_num} ===\n{body}\n=== END PAGE {page_num} ===")

    if plumber_pdf:
        plumber_pdf.close()

    if HAS_PDFPLUMBER and total_tables:
        print(f"[chorus-pdf] Table reconstruction: pdfplumber active ({total_tables} table(s) reconstructed)",
              file=sys.stderr)
    elif not HAS_PDFPLUMBER:
        print("[chorus-pdf] ⚠️  Table reconstruction skipped: pdfplumber unavailable — "
              "tables flattened to plain text (fallback, degraded output)",
              file=sys.stderr)

    final_text = "\n\n".join(parts)
    # Formula detection pass (§ "Formula strategy" — audit [#J] § F1/F2/F3):
    # tags Unicode math symbols, functional notation, and ambiguous flattened
    # exponents with a `[FORMULA: ...]` placeholder, mirroring the existing
    # FIGURE_PLACEHOLDER convention — never previously applied in text mode.
    final_text = detect_and_tag_formulas(final_text)
    n_formulas = final_text.count("[FORMULA:")

    with open(OUTPUT_PATH, "w", encoding="utf-8") as f:
        f.write(final_text)

    if n_formulas:
        print(f"[chorus-pdf]    {n_formulas} formula pattern(s) tagged [FORMULA: ...] — see § Formula strategy",
              file=sys.stderr)

    print(f"[chorus-pdf] ✅ {len(page_data)} pages extracted, {total_tables} table(s)", file=sys.stderr)
    if fig_pages:
        print(f"[chorus-pdf]    {fig_pages} page(s) contain figures — use --hybrid or --auto to extract them",
              file=sys.stderr)
    print(f"[chorus-pdf] Written to {OUTPUT_PATH}", file=sys.stderr)


if __name__ == "__main__":
    main()
```
